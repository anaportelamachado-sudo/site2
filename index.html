function controlarCriseDeAnsiedade() {
    // Técnica de respiração quadrada para desacelerar o sistema nervoso
    const ciclos = 4;
    
    for (let i = 1; i <= ciclos; i++) {
        console.log("Inhale o ar profundamente pelo nariz..."); // 4 segundos
        console.log("Prenda a respiração mantendo os pulmões cheios..."); // 4 segundos
        console.log("Exhale o ar lentamente pela boca..."); // 4 segundos
        console.log("Mantenha os pulmões vazios antes do próximo ciclo..."); // 4 segundos
    }
    
    return "Foco retornado ao presente com sucesso.";
}

def analisar_metricas_diarias(horas_trabalho, horas_sono, fez_pausa_lazer):
    # Analisa se a rotina atual infringe os limites saudáveis do organismo
    limite_trabalho_seguro = 8
    minimo_sono_necessario = 7
    
    if horas_trabalho > limite_trabalho_seguro or horas_sono < minimo_sono_necessario:
        if not fez_pausa_lazer:
            raise Exception("Risco crítico de Burnout detectado. Reduza os commits e descanse.")
    
    return "Estabilidade emocional operacional. Sistema equilibrado."
